#include <iostream> #Подключение библиотеки iostream

int main() #Начало функции main
{
    int a = 4, b = 5; #Присвоение начальных значений элементам a и b
    
    std::cout << "До обмена: a = " << a << ", b = " << b << std::endl; #Вывод значений a и b до обмена
    
    a = a + b;  # a=4+5=9
    b = a - b;  # b=9-5=4
    a = a - b;  # a=9-4=5
    
    std::cout << "После обмена: a = " << a << ", b = " << b << std::endl; #Вывод значений a и b после обмена
    
    return 0; #Конец функции main
}


Задание 2

#include <iostream>    #Подключение библиотеки iostream

int main()    #Начало функции main
{
    int n;   #Объявление переменной n
    std::cout << "Введите n: "; 
    std::cin >> n;  #Задаем значение n

    long long factorial = 1;  # Текущий факториал i!
    long long F = 1;          # Итоговое произведение 1!*2! *...*i!

    for (int i = 1; i <= n; i++) #Цикл работает пока i не станет равным n
    {
        factorial *= i;   # factorial становится равным i!
        F *= factorial;   # Домножаем F на i!
    }

    std::cout << "F(" << n << ") = " << F << std::endl; #Вывод значения F

    return 0;  #Конец функции main
}

#include <iostream> #Подключение библиотеки iostream

int main()  #Начало функции main
 {
    int A, B; #Объявление переменных A и B
    std::cout << "Введите A (0 или 1): ";  #Вывод строки "Введите A (0 или 1): "

    B = 2 - A;  #Замена значения B через арифметическое действие

    std::cout << "A = " << A << ", B = " << B << std::endl; #Вывод значений A и B

}
